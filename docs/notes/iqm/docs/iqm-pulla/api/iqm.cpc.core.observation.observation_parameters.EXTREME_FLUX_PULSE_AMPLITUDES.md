# EXTREME_FLUX_PULSE_AMPLITUDES

iqm.cpc.core.observation.observation_parameters.EXTREME_FLUX_PULSE_AMPLITUDES *= Parameter(name='extreme_pulse_amplitudes', label='Extreme flux pulse amplitudes', unit='', data_type=\<DataType.FLOAT: 1\>, collection_type=\<CollectionType.LIST: 1\>, element_indices=None)*
