Generic.kernel_vectors
     : forall (k n : nat) (y : Generic.BC.TA.cd k)
         (vectors : nat -> Generic.BC.TA.cd k),
       (forall j : nat,
        j < n -> Generic.BC.TA.mul k y (vectors j) = Generic.BC.TA.zero k) ->
       forall j : nat,
       j < n ->
       Generic.BC.TB.mul k (Generic.BC.tmap k y)
         (Generic.BC.tmap k (vectors j)) =
       Generic.BC.TB.zero k
Generic.kernel_spanned
     : forall (k n : nat) (y : Generic.BC.TA.cd k) 
         (d : IntegerRing.t)
         (back functionals vectors : nat -> Generic.BC.TA.cd k),
       (forall i : nat,
        i < 2 ^ k ->
        Generic.BC.TA.scale k d (Generic.BC.SA.basis k i) =
        Generic.BC.TA.add k
          (Generic.BC.source_map k back
             (Generic.BC.TA.mul k y (Generic.BC.SA.basis k i)))
          (Generic.source_comb k n functionals vectors
             (Generic.BC.SA.basis k i))) ->
       forall z : Generic.BC.TB.cd k,
       Generic.BC.TB.mul k (Generic.BC.tmap k y) z = Generic.BC.TB.zero k ->
       Generic.BC.TB.scale k (ZmodHom.h d) z =
       Generic.target_comb k n functionals vectors z
Generic.coefficients_recovered
     : forall (k n : nat) (d : IntegerRing.t)
         (functionals vectors : nat -> Generic.BC.TA.cd k),
       (forall j l : nat,
        j < n ->
        l < n ->
        Generic.source_pair k (functionals j) (vectors l) =
        (if j =? l then d else IntegerRing.zero)) ->
       forall (a : nat -> Zmod7.t) (l : nat),
       l < n ->
       Generic.target_pair k (Generic.BC.tmap k (functionals l))
         (Generic.BC.SB.tsum k n
            (fun j : nat =>
             Generic.BC.TB.scale k (a j) (Generic.BC.tmap k (vectors j)))) =
       Zmod7.mul (ZmodHom.h d) (a l)
Generic.kernel_free_basis
     : forall (k n : nat) (y : Generic.BC.TA.cd k) 
         (d : IntegerRing.t)
         (back functionals vectors : nat -> Generic.BC.TA.cd k),
       (forall j : nat,
        j < n -> Generic.BC.TA.mul k y (vectors j) = Generic.BC.TA.zero k) ->
       (forall i : nat,
        i < 2 ^ k ->
        Generic.BC.TA.scale k d (Generic.BC.SA.basis k i) =
        Generic.BC.TA.add k
          (Generic.BC.source_map k back
             (Generic.BC.TA.mul k y (Generic.BC.SA.basis k i)))
          (Generic.source_comb k n functionals vectors
             (Generic.BC.SA.basis k i))) ->
       forall inverse : Zmod7.t,
       Zmod7.mul (ZmodHom.h d) inverse = Zmod7.one ->
       forall z : Generic.BC.TB.cd k,
       Generic.BC.TB.mul k (Generic.BC.tmap k y) z = Generic.BC.TB.zero k <->
       z =
       Generic.BC.SB.tsum k n
         (fun j : nat =>
          Generic.BC.TB.scale k
            (Zmod7.mul inverse
               (Generic.target_pair k (Generic.BC.tmap k (functionals j)) z))
            (Generic.BC.tmap k (vectors j)))

         (functionals vectors : nat -> Generic.BC.TA.cd k),
       (forall j l : nat,
        j < n ->
        l < n ->
        Generic.source_pair k (functionals j) (vectors l) =
        (if j =? l then d else IntegerRing.zero)) ->
       forall inverse : Zmod7.t,
       Zmod7.mul (ZmodHom.h d) inverse = Zmod7.one ->
       forall (a : nat -> Zmod7.t) (l : nat),
       l < n ->
       a l =
       Zmod7.mul inverse
         (Generic.target_pair k (Generic.BC.tmap k (functionals l))
            (Generic.BC.SB.tsum k n
               (fun j : nat =>
                Generic.BC.TB.scale k (a j) (Generic.BC.tmap k (vectors j)))))
