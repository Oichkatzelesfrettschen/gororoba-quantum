"""Compare the circuits IBM received for every retained job, position by position, with the run's record.

Each retained run keeps its ISA circuits in isa.qpy, one circuit per PUB and
named by the PUB, and its record.json lists, per submitted job, the ordered
PUBs that job carried (``submission.jobs[].pubs``). The recovered wide-control
job keeps no record; its submitted-circuits.qpy holds the job's circuits in
submission order. The service returns a job's submitted program through
job.inputs: a QuantumProgram whose CircuitItem entries are the circuits the
executor ran, in order.

For every job of results/live/qpu-job-retention-audit.json this script
builds the expected ordered circuit list (isa.qpy circuits by the job's PUB
names, or the retained submission order) and records, per job, the expected
PUB names, the number of items the service returned, their item kinds, and
how many items equal the expected circuit at the same position under
QuantumCircuit equality (same DAG: gates, operands, parameters, global
phase).

``--write --out NEW --seed PREVIOUS`` needs the service and the account's
credential store. It validates and reuses the previous v3 rows, retrieves only
missing inputs, and publishes a new comparison exclusively after every row
checks. The successor binds the audit, seed, collector and replay versions;
existing comparison bytes remain immutable. ``--check --out NEW`` needs no network:
it requires the file to name exactly the audit's jobs and runs, every
retained circuit file to keep its recorded digest, each job's item count to
equal both its expected PUB list and the audit's pub_count, and the QPY
sha256 that --write recorded for every submitted item to equal the QPY
sha256 of the expected circuit at the same position, recomputed from the
retained file. A circuit the service returns and the equal archived circuit
serialize to identical QPY bytes, so the gate re-derives the comparison
rather than trusting a recorded count.
"""

from __future__ import annotations

import argparse
import hashlib
import importlib.metadata
import io
import json
import os
import sys
import tempfile
from datetime import UTC, datetime
from pathlib import Path
from typing import Any

ROOT = Path(__file__).resolve().parent.parent
sys.path.insert(0, str(ROOT / "gororoba_quantum"))
import ibm_account  # noqa: E402

AUDIT = ROOT / "results/live/qpu-job-retention-audit.json"
OUT = ROOT / "docs/evidence/submitted_circuits/comparison.json"
SCHEMA = "submitted-circuit-comparison-v3"
SUCCESSOR_SCHEMA = "submitted-circuit-comparison-v4"


def require(condition: bool, message: str) -> None:
    if not condition:
        raise SystemExit(message)


def audit_jobs() -> list[dict[str, Any]]:
    return list(json.loads(AUDIT.read_text(encoding="utf-8"))["jobs"])


def expected_order(job: dict[str, Any]) -> tuple[Path, list[str] | None]:
    """The retained circuit file and, for a recorded run, the job's ordered PUB names."""
    run = ROOT / job["run"]
    record = run / "record.json"
    if (run / "isa.qpy").is_file() and record.is_file():
        submission = json.loads(record.read_text(encoding="utf-8")).get("submission") or {}
        for entry in submission.get("jobs", []):
            if entry.get("job_id") == job["job_id"]:
                return run / "isa.qpy", [pub["name"] for pub in entry["pubs"]]
        raise SystemExit(f"{job['job_id']}: record.json lists no such submitted job")
    path = run / "submitted-circuits.qpy"
    require(path.is_file(), f"{job['run']} retains no circuit file")
    return path, None


def qpy_digest(circuit: Any) -> str:
    """sha256 of a circuit's QPY serialization; equal service and archived circuits serialize identically."""
    from qiskit import qpy

    buffer = io.BytesIO()
    qpy.dump(circuit, buffer)
    return hashlib.sha256(buffer.getvalue()).hexdigest()


def expected_circuits(path: Path, names: list[str] | None) -> list[Any]:
    """The retained circuits a job's items must equal, in order: the file's own order, or the named PUBs."""
    from qiskit import qpy

    with path.open("rb") as stream:
        retained = list(qpy.load(stream))
    if names is None:
        return retained
    by_name = {circuit.name: circuit for circuit in retained}
    require(all(name in by_name for name in names), f"{path}: a PUB name is missing")
    return [by_name[name] for name in names]


def write(output: Path = OUT, seed: Path | None = None) -> int:
    output = output.resolve()
    seed = seed.resolve() if seed is not None else None
    require(not output.exists(), "comparison output already exists; retain it and choose a new successor path")
    if seed is not None:
        seed.relative_to(ROOT)
    audited = audit_jobs()
    seeded: dict[str, dict[str, Any]] = {}
    if seed is not None:
        previous = json.loads(seed.read_text(encoding="utf-8"))
        require(previous.get("schema") == SCHEMA, "seed must be the retained v3 comparison")
        _check_rows(previous["jobs"], {job["job_id"]: job for job in audited}, complete=False)
        seeded = {row["job_id"]: row for row in previous["jobs"]}

    from qiskit_ibm_runtime import QiskitRuntimeService

    import live_run

    service = ibm_account.open_service(QiskitRuntimeService)
    rows = []
    retrieved = []
    for job in audited:
        if job["job_id"] in seeded:
            rows.append(seeded[job["job_id"]])
            continue
        path, names = expected_order(job)
        expected = expected_circuits(path, names)
        if names is None:
            names = [circuit.name for circuit in expected]
            source = "retained submission order"
        else:
            source = "record.json submission.jobs[].pubs"
        ibm_account.require_instance(
            ibm_account.recorded_instance(ROOT / job["run"]),
            live_run._instance_hash(service),
            f"{job['job_id']}: submitted-circuit lookup",
        )
        items = service.job(job["job_id"]).inputs["quantum_program"].items
        circuits = [getattr(item, "circuit", None) for item in items]
        submitted_digests = [qpy_digest(c) if c is not None else None for c in circuits]
        matched = sum(
            got is not None and index < len(expected) and got == expected[index] for index, got in enumerate(circuits)
        )
        rows.append(
            {
                "job_id": job["job_id"],
                "run": job["run"],
                "retained_file": path.relative_to(ROOT).as_posix(),
                "retained_sha256": hashlib.sha256(path.read_bytes()).hexdigest(),
                "order_source": source,
                "expected_pubs": names,
                "items": len(items),
                "item_kinds": sorted({type(item).__name__ for item in items}),
                "positional_matches": matched,
                "submitted_qpy_sha256": submitted_digests,
            }
        )
        retrieved.append(job["job_id"])
        print(json.dumps({k: v for k, v in rows[-1].items() if k != "expected_pubs"}), flush=True)
    _check_rows(rows, {job["job_id"]: job for job in audited})
    files = [AUDIT, ROOT / "scripts/compare_submitted_circuits.py"] + ([seed] if seed else [])
    document = {
        "schema": SUCCESSOR_SCHEMA,
        "jobs": rows,
        "provenance": {
            "retrieved_utc": datetime.now(UTC).isoformat(),
            "operation": "read existing job inputs; submissions and uploads excluded",
            "sources_sha256": {
                path.relative_to(ROOT).as_posix(): hashlib.sha256(path.read_bytes()).hexdigest() for path in files
            },
            "seed_file": seed.relative_to(ROOT).as_posix() if seed else None,
            "retrieved_job_ids": retrieved,
            "runtime_versions": {name: importlib.metadata.version(name) for name in ("qiskit", "qiskit-ibm-runtime")},
        },
    }
    _retain(output, document)
    return 0


def _retain(output: Path, document: dict[str, Any]) -> None:
    """Publish complete bytes exclusively, preserving every prior comparison."""
    output.parent.mkdir(parents=True, exist_ok=True)
    temporary = None
    try:
        with tempfile.NamedTemporaryFile(mode="w", encoding="utf-8", dir=output.parent, delete=False) as stream:
            temporary = Path(stream.name)
            stream.write(json.dumps(document, indent=1) + "\n")
            stream.flush()
            os.fsync(stream.fileno())
        os.link(temporary, output)
        directory = os.open(output.parent, os.O_RDONLY | os.O_DIRECTORY)
        try:
            os.fsync(directory)
        finally:
            os.close(directory)
    finally:
        if temporary is not None:
            temporary.unlink(missing_ok=True)


def _check_rows(rows: list[dict[str, Any]], audited: dict[str, Any], *, complete: bool = True) -> None:
    actual = {(row["job_id"], row["run"]) for row in rows}
    expected_identities = {(job_id, job["run"]) for job_id, job in audited.items()}
    require(actual == expected_identities if complete else actual <= expected_identities, "jobs differ")
    require(len(rows) == len(actual), "a job appears twice")
    for row in rows:
        # The file and PUB order come from the audited job's record, not from
        # the receipt, so a changed record fails even when the receipt follows it.
        path, names = expected_order(audited[row["job_id"]])
        require(path.relative_to(ROOT).as_posix() == row["retained_file"], f"{row['job_id']}: retained file differs")
        require(hashlib.sha256(path.read_bytes()).hexdigest() == row["retained_sha256"], f"{path} changed")
        circuits = expected_circuits(path, names)
        source = "retained submission order" if names is None else "record.json submission.jobs[].pubs"
        require(row["order_source"] == source, f"{row['job_id']}: order source differs from the record")
        require(
            row["expected_pubs"] == [c.name for c in circuits], f"{row['job_id']}: PUB order differs from the record"
        )
        pub_count = audited[row["job_id"]]["pub_count"]
        counts_agree = row["items"] == pub_count == len(row["expected_pubs"])
        require(counts_agree, f"{row['job_id']}: item and PUB counts differ")
        expected = [qpy_digest(circuit) for circuit in circuits]
        require(len(row["submitted_qpy_sha256"]) == row["items"], f"{row['job_id']}: item digests are incomplete")
        mismatched = [
            k for k, (got, want) in enumerate(zip(row["submitted_qpy_sha256"], expected, strict=True)) if got != want
        ]
        require(not mismatched, f"{row['job_id']}: submitted items {mismatched} differ from the retained circuits")
        require(row["positional_matches"] == row["items"], f"{row['job_id']}: an item differs from its circuit")
        require(row["item_kinds"] == ["CircuitItem"], f"{row['job_id']}: item kinds {row['item_kinds']}")


def check(output: Path = OUT) -> int:
    data = json.loads(output.read_text(encoding="utf-8"))
    require(data.get("schema") in {SCHEMA, SUCCESSOR_SCHEMA}, "unexpected schema")
    rows = data["jobs"]
    _check_rows(rows, {job["job_id"]: job for job in audit_jobs()})
    if data["schema"] == SUCCESSOR_SCHEMA:
        provenance = data["provenance"]
        seed_name = provenance["seed_file"]
        required_sources = {AUDIT.relative_to(ROOT).as_posix(), "scripts/compare_submitted_circuits.py"}
        if seed_name is not None:
            require(isinstance(seed_name, str), "seed source path must be a string")
            (ROOT / seed_name).resolve().relative_to(ROOT)
            required_sources.add(seed_name)
        require(set(provenance["sources_sha256"]) == required_sources, "provenance source identities differ")
        for name, digest in provenance["sources_sha256"].items():
            require(
                hashlib.sha256((ROOT / name).read_bytes()).hexdigest() == digest, f"{name}: provenance source changed"
            )
        require(
            set(provenance["runtime_versions"]) == {"qiskit", "qiskit-ibm-runtime"},
            "provenance runtime identities differ",
        )
        for name, version in provenance["runtime_versions"].items():
            require(importlib.metadata.version(name) == version, f"{name}: replay runtime differs")
        previous = json.loads((ROOT / seed_name).read_text()) if seed_name else {"jobs": []}
        seeded = {row["job_id"]: row for row in previous["jobs"]}
        require(all(row == seeded[row["job_id"]] for row in rows if row["job_id"] in seeded), "seeded rows changed")
        missing = {row["job_id"] for row in rows} - seeded.keys()
        retrieved = provenance["retrieved_job_ids"]
        require(set(retrieved) == missing and len(retrieved) == len(missing), "retrieved job identities differ")
    items = sum(row["items"] for row in rows)
    print(f"submitted circuits: {len(rows)} jobs, {items} items, each equal to its expected circuit in order")
    return 0


def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__.splitlines()[0])
    mode = parser.add_mutually_exclusive_group(required=True)
    mode.add_argument("--write", action="store_true")
    mode.add_argument("--check", action="store_true")
    parser.add_argument(
        "--out", type=Path, default=OUT, help="comparison to retain or check; existing bytes are immutable"
    )
    parser.add_argument(
        "--seed", type=Path, help="validated v3 comparison whose job rows are reused without service reads"
    )
    args = parser.parse_args()
    if args.seed is not None and not args.write:
        parser.error("--seed requires --write")
    return write(args.out, args.seed) if args.write else check(args.out)


if __name__ == "__main__":
    sys.exit(main())
