@lru_cache(None)
def mul(d: int, i: int, j: int) -> tuple[int, int]:
    if d < 1 or d & (d-1) or not (0 <= i < d and 0 <= j < d):
        raise ValueError('dimension must be a power of two; indices must be in range')
    if d == 1: return 1, 0
    h = d // 2
    if i < h and j < h: return mul(h,i,j)
    if i < h <= j:
        s,k=mul(h,j-h,i); return s,k+h
    if j < h <= i:
        s,k=mul(h,i-h,j); return s*(1 if j==0 else -1),k+h
    s,k=mul(h,j-h,i-h); return -s*(1 if j==h else -1),k

def anf(values: list[int]) -> list[int]:
    a=values.copy(); d=len(a)
    if d == 0 or d & (d-1): raise ValueError('power-of-two truth table required')
    bit=1
    while bit < d:
        for x in range(d):
            if x&bit: a[x]^=a[x^bit]
        bit <<= 1
    return [m for m,v in enumerate(a) if v]

def fwht(values: list[int]) -> list[int]:
    a=values.copy(); step=1
    while step < len(a):
        for start in range(0,len(a),2*step):
            for k in range(step):
                u,v=a[start+k],a[start+k+step]
                a[start+k],a[start+k+step]=u+v,u-v
        step *= 2
    return a
