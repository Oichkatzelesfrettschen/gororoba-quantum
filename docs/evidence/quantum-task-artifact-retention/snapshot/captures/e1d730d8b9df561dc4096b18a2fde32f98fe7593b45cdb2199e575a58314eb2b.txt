"""Render retained raw frequencies without inferring interval coverage."""

import hashlib
import json
from pathlib import Path

import matplotlib

matplotlib.use("Agg")
import matplotlib.pyplot as plt

root = Path(__file__).resolve().parents[3]
acquisition = root / "results/quantum_inspire/runs/tuna17-quaternion-order-noise"
summary_path = acquisition / "pilot-summary.json"
summary = json.loads(summary_path.read_text())
for name, digest in summary["analysis_files_sha256"].items():
    if hashlib.sha256((acquisition / name).read_bytes()).hexdigest() != digest:
        raise ValueError("retained analysis binding differs")
settings = summary["settings"]
positions = list(range(len(settings)))
labels = [
    f"{'0' if row['setting']['theta'] == 0 else 'pi/4'}, "
    f"{'0' if row['setting']['phi'] == 0 else 'pi/4'}\n"
    f"{'identity' if row['setting']['representative'] == 0 else 'right-j'}"
    for row in settings
]
figure, axis = plt.subplots(figsize=(11, 5.8), layout="constrained")
axis.scatter(positions, [row["observed_dark_probability"] for row in settings], s=70, color="#2455b5", label="Measured raw frequency")
axis.scatter(positions, [row["native_dark_probability"] for row in settings], marker="_", s=300, linewidth=2.5, color="#b34b16", label="Ideal native prediction")
axis.set(xticks=positions, xticklabels=labels, ylim=(-0.015, 0.32), ylabel="Grouped dark-port probability", xlabel="i phase, j phase; global quaternion representative")
axis.set_title("Tuna-17 complex-encoded quaternion order pilot", loc="left", fontweight="bold")
axis.grid(axis="y", alpha=0.25)
axis.legend(loc="upper left")
figure.suptitle("24 completed jobs; 393,216 shots; 16,384 shots per job", fontsize=11, x=0.5)
figure.supxlabel("Fixed chronological order and one acquisition per setting confound drift and circuit effects.\nRaw frequencies establish neither representative equivalence nor fundamental quaternionic physics.", fontsize=9)
destination = Path(__file__).parent / "tuna-order-pilot-probabilities.png"
figure.savefig(destination, dpi=170)
plt.close(figure)
metadata = {"summary_sha256": hashlib.sha256(summary_path.read_bytes()).hexdigest(), "producer_sha256": hashlib.sha256(Path(__file__).read_bytes()).hexdigest(), "png_sha256": hashlib.sha256(destination.read_bytes()).hexdigest(), "interpretation": "raw measured frequencies and ideal predictions; confidence coverage unestablished"}
destination.with_suffix(".json").write_text(json.dumps(metadata, indent=2) + "\n")
